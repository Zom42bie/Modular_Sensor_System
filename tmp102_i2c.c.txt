// File: tmp102_i2c.c
#include "stm32f4xx_hal.h"
#include <stdint.h>

// TMP102 I2C address and register
#define TMP102_ADDR (0x48 << 1) // 7-bit address shifted left
#define TEMP_REG 0x00

I2C_HandleTypeDef hi2c1;

// Initialize I2C for TMP102
void TMP102_Init(I2C_HandleTypeDef *hi2c) {
    hi2c1 = *hi2c;
    // Configure I2C (assumes STM32CubeMX setup for I2C1)
    // Example: 100 kHz, 7-bit addressing
}

// Read temperature from TMP102 (returns fixed-point value)
int16_t TMP102_ReadTemp(void) {
    uint8_t data[2];
    HAL_StatusTypeDef status;

    // Read 2 bytes from temperature register
    status = HAL_I2C_Mem_Read(&hi2c1, TMP102_ADDR, TEMP_REG, 1, data, 2, 100);
    if (status != HAL_OK) return 0; // Error handling

    // Convert to 12-bit temperature (fixed-point, 0.0625°C resolution)
    int16_t raw = (data[0] << 4) | (data[1] >> 4);
    if (raw & 0x800) raw |= 0xF000; // Sign extend for negative temps
    return raw; // Fixed-point: 1 unit = 0.0625°C
}
